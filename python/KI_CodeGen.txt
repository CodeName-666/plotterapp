import threading
import serial
import can

# Serial reader class
class SerialReader(threading.Thread):
    def __init__(self, serial_port, baud_rate):
        # Set up the serial port
        self.ser = serial.Serial(serial_port, baud_rate)
        # Set up the thread
        threading.Thread.__init__(self)

    def run(self):
        # Continuously read data from the serial port
        while True:
            data = self.ser.readline()
            print(data)

# CAN reader class
class CANReader(threading.Thread):
    def __init__(self, can_interface):
        # Set up the CAN interface
        self.bus = can.interface.Bus(can_interface, bustype="socketcan")
        # Set up the thread
        threading.Thread.__init__(self)

    def run(self):
        # Continuously read data from the CAN bus
        while True:
            message = self.bus.recv()
            print(message)

# Data receiver class
class DataReceiver(threading.Thread):
    def __init__(self, protocol, reader_class, **kwargs):
        # Set up the communication protocol
        if protocol == "serial":
            # Set up the serial reader
            self.reader = reader_class(kwargs["serial_port"], kwargs["baud_rate"])
        elif protocol == "can":
            # Set up the CAN reader
            self.reader = reader_class(kwargs["can_interface"])
        else:
            raise ValueError("Invalid protocol specified: {}".format(protocol))
        # Set up the thread
        threading.Thread.__init__(self)

    def run(self):
        # Start the reader thread
        self.reader.start()

# Create a new DataReceiver thread that uses the serial protocol
thread1 = DataReceiver("serial", SerialReader, serial_port="/dev/ttyS0", baud_rate=9600)

# Create a new DataReceiver thread that uses the CAN protocol
thread2 = DataReceiver("can", CANReader, can_interface="can0")

# Start the threads
thread1.start()
thread2.start()
